/* =========================================================================
   3. WEBGL CINEMATIC DARK WAVE BACKGROUND
   ========================================================================= */

const BackgroundCanvas: React.FC<{ opacity?: number }> = ({ opacity = 1 }) => {
  const canvasRef = useRef<HTMLCanvasElement | null>(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;

    const gl = canvas.getContext('webgl');
    if (!gl) return;

    const vsSource = `
      attribute vec2 p;
      void main() {
        gl_Position = vec4(p, 0.0, 1.0);
      }
    `;

    const fsSource = `
      precision highp float;
      uniform float t;
      uniform vec2 r;
      void main() {
        vec2 p = (gl_FragCoord.xy * 2.0 - r) / min(r.x, r.y);
        for (float i = 1.0; i < 8.0; i++) {
          p.x += 0.3 / i * sin(i * 3.0 * p.y + t);
          p.y += 0.3 / i * cos(i * 3.0 * p.x + t);
        }
        float c = 0.5 + 0.5 * sin(p.x + p.y);
        gl_FragColor = vec4(vec3(c * 0.12 + 0.03) + pow(c, 12.0) * 0.3, 1.0);
      }
    `;

    const createShader = (type: number, source: string) => {
      const shader = gl.createShader(type);
      if (!shader) return null;
      gl.shaderSource(shader, source);
      gl.compileShader(shader);
      if (!gl.getShaderParameter(shader, gl.COMPILE_STATUS)) {
        gl.deleteShader(shader);
        return null;
      }
      return shader;
    };

    const vertShader = createShader(gl.VERTEX_SHADER, vsSource);
    const fragShader = createShader(gl.FRAGMENT_SHADER, fsSource);
    if (!vertShader || !fragShader) return;

    const program = gl.createProgram();
    if (!program) return;

    gl.attachShader(program, vertShader);
    gl.attachShader(program, fragShader);
    gl.linkProgram(program);

    if (!gl.getProgramParameter(program, gl.LINK_STATUS)) return;
    gl.useProgram(program);

    const positionBuffer = gl.createBuffer();
    gl.bindBuffer(gl.ARRAY_BUFFER, positionBuffer);
    gl.bufferData(
      gl.ARRAY_BUFFER,
      new Float32Array([-1, -1, 1, -1, -1, 1, 1, 1]),
      gl.STATIC_DRAW
    );

    const posAttrLoc = gl.getAttribLocation(program, 'p');
    gl.enableVertexAttribArray(posAttrLoc);
    gl.vertexAttribPointer(posAttrLoc, 2, gl.FLOAT, false, 0, 0);

    const timeUniformLoc = gl.getUniformLocation(program, 't');
    const resUniformLoc = gl.getUniformLocation(program, 'r');

    let animationFrameId: number;

    const handleResize = () => {
      if (!canvas) return;
      canvas.width = window.innerWidth;
      canvas.height = window.innerHeight;
      gl.viewport(0, 0, canvas.width, canvas.height);
    };

    window.addEventListener('resize', handleResize);
    handleResize();

    const render = (time: number) => {
      if (resUniformLoc) gl.uniform2f(resUniformLoc, canvas.width, canvas.height);
      if (timeUniformLoc) gl.uniform1f(timeUniformLoc, time * 0.0015);
      gl.drawArrays(gl.TRIANGLE_STRIP, 0, 4);
      animationFrameId = requestAnimationFrame(render);
    };

    animationFrameId = requestAnimationFrame(render);

    return () => {
      window.removeEventListener('resize', handleResize);
      cancelAnimationFrame(animationFrameId);
      gl.deleteProgram(program);
      gl.deleteShader(vertShader);
      gl.deleteShader(fragShader);
      gl.deleteBuffer(positionBuffer);
    };
  }, []);

  return (
    <canvas
      ref={canvasRef}
      className="absolute inset-0 w-full h-full pointer-events-none transition-opacity duration-1000"
      style={{ opacity }}
    />
  );
};

/* =========================================================================
   4. REALISTIC RAIN CANVAS (with Splashes)
   ========================================================================= */

interface RainDrop {
  x: number;
  y: number;
  length: number;
  speed: number;
  thickness: number;
  alpha: number;
  layer: 'fore' | 'mid' | 'back';
}

interface Splash {
  x: number;
  y: number;
  vx: number;
  vy: number;
  alpha: number;
  radius: number;
}

const RainCanvas: React.FC<{
  density?: number;
  windAngle?: number;
  speedMultiplier?: number;
  opacity?: number;
}> = ({ density = 160, windAngle = 0.12, speedMultiplier = 1.05, opacity = 0.85 }) => {
  const canvasRef = useRef<HTMLCanvasElement | null>(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    let width = (canvas.width = window.innerWidth);
    let height = (canvas.height = window.innerHeight);

    const handleResize = () => {
      if (!canvas) return;
      width = canvas.width = window.innerWidth;
      height = canvas.height = window.innerHeight;
    };

    window.addEventListener('resize', handleResize);

    const drops: RainDrop[] = [];
    const count = Math.floor(density * (width / 1200));

    for (let i = 0; i < count; i++) {
      const isFore = Math.random() > 0.65;
      const isMid = !isFore && Math.random() > 0.4;
      drops.push({
        x: Math.random() * (width + 200) - 100,
        y: Math.random() * height,
        length: (isFore ? 26 : isMid ? 18 : 12) + Math.random() * 8,
        speed: ((isFore ? 20 : isMid ? 14 : 9) + Math.random() * 6) * speedMultiplier,
        thickness: isFore ? 1.6 : isMid ? 1.2 : 0.8,
        alpha: isFore ? 0.65 : isMid ? 0.45 : 0.25,
        layer: isFore ? 'fore' : isMid ? 'mid' : 'back',
      });
    }

    const splashes: Splash[] = [];
    let animationId: number;
    const windX = Math.sin(windAngle);
    const windY = Math.cos(windAngle);

    const animate = () => {
      ctx.clearRect(0, 0, width, height);

      for (let i = 0; i < drops.length; i++) {
        const d = drops[i];
        ctx.beginPath();
        ctx.moveTo(d.x, d.y);
        ctx.lineTo(d.x + windX * d.length, d.y + windY * d.length);
        ctx.strokeStyle = `rgba(215, 230, 255, ${d.alpha * opacity})`;
        ctx.lineWidth = d.thickness;
        ctx.lineCap = 'round';
        ctx.stroke();

        d.x += windX * d.speed;
        d.y += windY * d.speed;

        if (d.y > height) {
          if (d.layer === 'fore' && Math.random() > 0.6) {
            splashes.push({
              x: d.x,
              y: height - Math.random() * 10,
              vx: (Math.random() - 0.5) * 3,
              vy: -(Math.random() * 2 + 1),
              alpha: 0.6,
              radius: Math.random() * 1.5 + 0.8,
            });
          }
          d.y = -d.length - Math.random() * 40;
          d.x = Math.random() * (width + 300) - 150;
        }

        if (d.x > width + 150) d.x = -100;
        else if (d.x < -150) d.x = width + 100;
      }

      for (let s = splashes.length - 1; s >= 0; s--) {
        const sp = splashes[s];
        ctx.beginPath();
        ctx.arc(sp.x, sp.y, sp.radius, 0, Math.PI * 2);
        ctx.fillStyle = `rgba(220, 235, 255, ${sp.alpha * opacity})`;
        ctx.fill();

        sp.x += sp.vx;
        sp.y += sp.vy;
        sp.vy += 0.15;
        sp.alpha *= 0.9;

        if (sp.alpha < 0.05 || sp.y > height) splashes.splice(s, 1);
      }

      animationId = requestAnimationFrame(animate);
    };

    animationId = requestAnimationFrame(animate);

    return () => {
      window.removeEventListener('resize', handleResize);
      cancelAnimationFrame(animationId);
    };
  }, [density, windAngle, speedMultiplier, opacity]);

  return (
    <canvas
      ref={canvasRef}
      className="absolute inset-0 w-full h-full pointer-events-none z-20"
      style={{ mixBlendMode: 'screen' }}
    />
  );
};

/* =========================================================================
   5. REALISTIC SNOW CANVAS (Multi-Plane Parallax Depth)
   ========================================================================= */

interface SnowflakeItem {
  x: number;
  y: number;
  radius: number;
  speed: number;
  drift: number;
  phase: number;
  phaseSpeed: number;
  alpha: number;
  layer: 'fore' | 'mid' | 'back';
}

const SnowCanvas: React.FC<{
  density?: number;
  speedMultiplier?: number;
  opacity?: number;
}> = ({ density = 140, speedMultiplier = 1.0, opacity = 0.9 }) => {
  const canvasRef = useRef<HTMLCanvasElement | null>(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    let width = (canvas.width = window.innerWidth);
    let height = (canvas.height = window.innerHeight);

    const handleResize = () => {
      if (!canvas) return;
      width = canvas.width = window.innerWidth;
      height = canvas.height = window.innerHeight;
    };

    window.addEventListener('resize', handleResize);

    const flakes: SnowflakeItem[] = [];
    const count = Math.floor(density * (width / 1200));

    for (let i = 0; i < count; i++) {
      const rand = Math.random();
      let layer: 'fore' | 'mid' | 'back' = 'mid';
      let radius = 1.6 + Math.random() * 1.5;
      let speed = (1.0 + Math.random() * 0.9) * speedMultiplier;
      let alpha = 0.45 + Math.random() * 0.3;

      if (rand > 0.8) {
        layer = 'fore';
        radius = 3.0 + Math.random() * 2.5;
        speed = (1.8 + Math.random() * 1.4) * speedMultiplier;
        alpha = 0.75 + Math.random() * 0.25;
      } else if (rand < 0.35) {
        layer = 'back';
        radius = 0.8 + Math.random() * 1.0;
        speed = (0.5 + Math.random() * 0.6) * speedMultiplier;
        alpha = 0.2 + Math.random() * 0.25;
      }

      flakes.push({
        x: Math.random() * width,
        y: Math.random() * height,
        radius,
        speed,
        drift: (Math.random() - 0.5) * 0.8,
        phase: Math.random() * Math.PI * 2,
        phaseSpeed: 0.01 + Math.random() * 0.02,
        alpha,
        layer,
      });
    }

    let animationFrameId: number;

    const render = () => {
      ctx.clearRect(0, 0, width, height);

      for (let i = 0; i < flakes.length; i++) {
        const flake = flakes[i];
        flake.phase += flake.phaseSpeed;
        const swayX = Math.sin(flake.phase) * (flake.layer === 'fore' ? 1.4 : 0.8);

        flake.x += flake.drift + swayX;
        flake.y += flake.speed;

        if (flake.y > height + 10) {
          flake.y = -10;
          flake.x = Math.random() * width;
        }
        if (flake.x > width + 10) flake.x = -10;
        else if (flake.x < -10) flake.x = width + 10;

        ctx.save();
        ctx.beginPath();
        ctx.arc(flake.x, flake.y, flake.radius, 0, Math.PI * 2);

        if (flake.layer === 'fore') {
          const grad = ctx.createRadialGradient(
            flake.x,
            flake.y,
            0,
            flake.x,
            flake.y,
            flake.radius * 1.6
          );
          grad.addColorStop(0, `rgba(255, 255, 255, ${flake.alpha * opacity})`);
          grad.addColorStop(0.6, `rgba(240, 248, 255, ${flake.alpha * 0.6 * opacity})`);
          grad.addColorStop(1, 'rgba(255, 255, 255, 0)');
          ctx.fillStyle = grad;
        } else {
          ctx.fillStyle = `rgba(245, 250, 255, ${flake.alpha * opacity})`;
        }

        ctx.fill();
        ctx.restore();
      }

      animationFrameId = requestAnimationFrame(render);
    };

    animationFrameId = requestAnimationFrame(render);

    return () => {
      window.removeEventListener('resize', handleResize);
      cancelAnimationFrame(animationFrameId);
    };
  }, [density, speedMultiplier, opacity]);

  return (
    <canvas
      ref={canvasRef}
      className="absolute inset-0 pointer-events-none z-10"
      style={{ mixBlendMode: 'screen' }}
    />
  );
};

const GlowButton: React.FC<{
  id: string;
  label: string;
  icon?: React.ReactNode;
  accentGlowColor?: string;
  className?: string;
  offsetClass?: string;
  onClick: () => void;
}> = ({
  id,
  label,
  icon,
  accentGlowColor = 'rgba(255, 255, 255,',
  className = '',
  offsetClass = '',
  onClick,
}) => {
  const btnRef = useRef<HTMLButtonElement | null>(null);

  const handleMouseMove = (e: React.MouseEvent<HTMLButtonElement>) => {
    const btn = btnRef.current;
    if (!btn) return;
    const rect = btn.getBoundingClientRect();
    const x = e.clientX - rect.left;
    const y = e.clientY - rect.top;
    btn.style.setProperty('--x', `${x}px`);
    btn.style.setProperty('--y', `${y}px`);

    const centerX = rect.width / 2;
    const centerY = rect.height / 2;
    const dist = Math.sqrt(Math.pow(x - centerX, 2) + Math.pow(y - centerY, 2));
    const maxDist = Math.sqrt(Math.pow(centerX, 2) + Math.pow(centerY, 2));
    const intensity = 0.15 + (dist / maxDist) * 0.65;
    btn.style.setProperty('--i', intensity.toString());
  };

  const handleMouseLeave = () => {
    const btn = btnRef.current;
    if (!btn) return;
    btn.style.setProperty('--i', '0');
  };

  return (
    <div className={`relative flex flex-col items-center ${offsetClass}`}>
      <button
        id={id}
        ref={btnRef}
        type="button"
        onClick={onClick}
        onMouseMove={handleMouseMove}
        onMouseLeave={handleMouseLeave}
        className={`relative flex items-center justify-center gap-3.5 px-8 sm:px-10 py-4.5 text-[18px] md:text-[19px] font-black tracking-[4px] text-white rounded-2xl cursor-pointer overflow-hidden transition-all duration-300 hover:scale-105 active:scale-95 select-none ${className}`}
        style={{
          background: 'rgba(255, 255, 255, 0.04)',
          backdropFilter: 'blur(25px)',
          WebkitBackdropFilter: 'blur(25px)',
          border: '1px solid rgba(255, 255, 255, 0.15)',
          boxShadow: '0 8px 32px 0 rgba(0, 0, 0, 0.45)',
        }}
      >
        <div
          className="absolute inset-0 pointer-events-none transition-opacity duration-150"
          style={{
            background: `radial-gradient(circle at var(--x, 50%) var(--y, 50%), ${accentGlowColor} var(--i, 0)) 0%, transparent 75%)`,
          }}
        />
        <div className="relative z-10 flex items-center gap-3 pointer-events-none">
          {icon && <span className="opacity-90">{icon}</span>}
          <span className="drop-shadow-[0_2px_8px_rgba(255,255,255,0.4)]">{label}</span>
        </div>
      </button>
    </div>
  );
};

      {/* WebGL Dark Wave Background */}
      <BackgroundCanvas opacity={stage === 'main' ? 1 : 0.3} />

      {/* Rain Canvas */}
      {weather === 'rain' && (
        <RainCanvas
          density={170}
          windAngle={0.12}
          speedMultiplier={1.05}
          opacity={0.85}
        />
      )}

      {/* Snow Canvas */}
      {weather === 'snow' && (
        <SnowCanvas density={150} speedMultiplier={1.0} opacity={0.9} />
      )}

      {/* Header Controls */}
      {stage === 'main' && (
        <header className="fixed top-0 left-0 right-0 z-40 flex items-center justify-between px-6 py-5 pointer-events-none">
          <div className="text-xs uppercase tracking-[0.4em] font-mono text-neutral-500 pointer-events-auto">
            Localx • Portal
          </div>

          <div className="flex items-center gap-2.5 pointer-events-auto">
            <button
              id="image-settings-btn"
              type="button"
              onClick={() => setIsImageModalOpen(true)}
              title="이미지 URL 직접 설정"
              className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-xs font-semibold bg-white/5 hover:bg-white/15 border border-white/10 text-neutral-300 hover:text-white transition-all backdrop-blur-md cursor-pointer"
            >
              <ImageIcon className="w-3.5 h-3.5 text-neutral-300" />
              <span className="hidden sm:inline">이미지 설정</span>
            </button>

            {/* Weather Selector */}
            <div
              id="weather-selector"
              className="flex items-center rounded-xl bg-white/5 border border-white/10 p-0.5 backdrop-blur-md shadow-sm"
            >
              <button
                id="weather-rain-btn"
                type="button"
                onClick={() => {
                  setWeather('rain');
                  showToast({ title: '비 효과가 켜졌습니다 🌧️', type: 'info' });
                }}
                title="비 효과 (Rain)"
                className={`flex items-center gap-1 px-2.5 py-1.5 rounded-lg text-xs font-semibold transition-all cursor-pointer ${
                  weather === 'rain'
                    ? 'bg-cyan-500/20 text-cyan-300 border border-cyan-500/35 shadow-[0_0_12px_rgba(6,182,212,0.25)]'
                    : 'text-neutral-400 hover:text-white hover:bg-white/5 border border-transparent'
                }`}
              >
                <CloudRain className="w-3.5 h-3.5" />
                <span className="hidden sm:inline">비</span>
              </button>

              <button
                id="weather-snow-btn"
                type="button"
                onClick={() => {
                  setWeather('snow');
                  showToast({ title: '눈 효과가 켜졌습니다 ❄️', type: 'info' });
                }}
                title="눈 효과 (Snow)"
                className={`flex items-center gap-1 px-2.5 py-1.5 rounded-lg text-xs font-semibold transition-all cursor-pointer ${
                  weather === 'snow'
                    ? 'bg-sky-400/20 text-sky-200 border border-sky-400/35 shadow-[0_0_12px_rgba(56,189,248,0.25)]'
                    : 'text-neutral-400 hover:text-white hover:bg-white/5 border border-transparent'
                }`}
              >
                <Snowflake className="w-3.5 h-3.5" />
                <span className="hidden sm:inline">눈</span>
              </button>

              <button
                id="weather-off-btn"
                type="button"
                onClick={() => {
                  setWeather('none');
                  showToast({ title: '날씨 효과가 꺼졌습니다', type: 'info' });
                }}
                title="날씨 끄기"
                className={`p-1.5 rounded-lg text-xs transition-all cursor-pointer ${
                  weather === 'none'
                    ? 'bg-white/15 text-white border border-white/20'
                    : 'text-neutral-500 hover:text-neutral-300 hover:bg-white/5 border border-transparent'
                }`}
              >
                <CloudOff className="w-3.5 h-3.5" />
              </button>
            </div>

            <button
              id="replay-btn"
              type="button"
              onClick={handleReplay}
              title="인트로 애니메이션 다시보기"
              className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-xs font-semibold bg-white/5 hover:bg-white/15 border border-white/10 text-neutral-300 hover:text-white transition-all backdrop-blur-md cursor-pointer"
            >
              <RotateCcw className="w-3.5 h-3.5" />
              <span className="hidden sm:inline">다시보기</span>
            </button>
          </div>
        </header>
      )}

      {/* Main Content Dashboard */}
      {stage === 'main' && (
        <main
          id="portal-canvas"
          className="relative z-10 w-full h-full flex flex-col justify-between items-center py-10 px-4"
        >
          <motion.div
            initial={{ opacity: 0, y: -16 }}
            animate={{ opacity: 1, y: 0 }}
            transition={{ duration: 0.7, ease: 'easeOut' }}
            className="pt-4 sm:pt-8 text-center"
          >
            <h1
              id="top-title"
              className="text-sm sm:text-base md:text-lg font-black tracking-[10px] sm:tracking-[12px] uppercase text-white/70 select-none"
            >
              Localx on top
            </h1>
          </motion.div>

          {/* Center Entity: Fixed Size Gentle Float */}
          <div className="relative my-auto flex flex-col items-center justify-center">
            <motion.div
              id="central-floating-l"
              animate={{
                y: [-9, 9, -9],
              }}
              transition={{
                duration: 4.6,
                repeat: Infinity,
                ease: 'easeInOut',
              }}
              className="relative z-20 flex flex-col items-center justify-center cursor-default select-none"
            >
              <img
                src={displayMainLogo}
                alt="Localx L Logo"
                className="w-40 h-40 sm:w-48 sm:h-48 md:w-56 md:h-56 object-contain rounded-2xl pointer-events-none filter drop-shadow-[0_0_35px_rgba(255,255,255,0.45)]"
              />
            </motion.div>

            <motion.div
              animate={{
                opacity: [0.35, 0.15, 0.35],
                scaleX: [1.05, 0.85, 1.05],
              }}
              transition={{
                duration: 4.6,
                repeat: Infinity,
                ease: 'easeInOut',
              }}
              className="w-28 sm:w-36 h-3 rounded-full bg-white/20 blur-md mt-4 pointer-events-none"
            />
          </div>

          {/* Social Action Buttons: YouTube & Discord */}
          <motion.div
            initial={{ opacity: 0, y: 24 }}
            animate={{ opacity: 1, y: 0 }}
            transition={{ duration: 0.7, delay: 0.15, ease: 'easeOut' }}
            className="w-full flex flex-col sm:flex-row items-center justify-center gap-6 sm:gap-8 px-4 pb-4"
          >
            <GlowButton
              id="yt-btn"
              label="YOUTUBE"
              icon={<Youtube className="w-5 h-5 text-red-500" />}
              accentGlowColor="rgba(255, 60, 60,"
              className="min-w-[210px] sm:min-w-[230px] border-white/20 hover:border-red-500/50"
              offsetClass="sm:-translate-x-3"
              onClick={() => setActivePlatform('youtube')}
            />

            <GlowButton
              id="discord-btn"
              label="DISCORD"
              icon={<MessageSquare className="w-5 h-5 text-indigo-400" />}
              accentGlowColor="rgba(114, 137, 218,"
              className="min-w-[210px] sm:min-w-[230px] border-white/20 hover:border-indigo-400/50"
              onClick={() => setActivePlatform('discord')}
            />
          </motion.div>
        </main>
      )}

      {/* 🎵 YOUTUBE MUSIC PLAYER (Draggable + Continuous Background Audio) */}
      {stage === 'main' && (
        <YouTubePlayer
          videoId={assets.youtubeMusicId || 'tm9GpM3lV3s'}
          defaultExpanded={true}
        />
      )}

      {/* FULL-SCREEN PLATFORM TRANSITION */}
      <PlatformScreen
        platform={activePlatform}
        onClose={() => setActivePlatform(null)}
        onCopySuccess={(msg) =>
          showToast({
            title: msg,
            type: 'success',
          })
        }
      />

      {/* Toast Notification */}
      <Toast toast={toast} onClose={() => setToast(null)} />

      {/* Image Settings Modal */}
      <ImageSettingsModal
        isOpen={isImageModalOpen}
        onClose={() => setIsImageModalOpen(false)}
        config={assets}
        onSave={handleSaveAssets}
        onReset={handleResetAssets}
      />
    </div>
  );
}
